def binary(n):
    return format(n & 0xFF, '08b')

def add(a, b):
    A = binary(a)
    B = binary(b)
    result = (a + b) & 0xFFFF

    print("First number :", A)
    print("Second number:", B)
    print("Result (16-bit):", format(result, '016b'))
    print("Decimal result:", a + b)
    print("Overflow:", "Yes" if not -32768 <= a + b <= 32767 else "No")
    print()

print("a) 24 + 17")
add(24, 17)

print("b) -25 + (-18)")
add(-25, -18)
